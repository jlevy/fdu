/* Peak RSS of one command, read with wait4 from a launcher small enough that the
 * kernel's inherited high-water mark (exec records the old mm's) is negligible.
 * usage: rss COMMAND [ARGS...]   prints "<maxrss_kib> <exit_status>" on stderr. */
#include <stdio.h>
#include <stdlib.h>
#include <sys/resource.h>
#include <sys/wait.h>
#include <unistd.h>
#include <fcntl.h>

int main(int argc, char **argv) {
    if (argc < 2) return 2;
    pid_t pid = fork();
    if (pid < 0) return 2;
    if (pid == 0) {
        int devnull = open("/dev/null", O_WRONLY);
        if (devnull >= 0) { dup2(devnull, 1); }
        execv(argv[1], argv + 1);
        _exit(127);
    }
    int status = 0;
    struct rusage usage;
    if (wait4(pid, &status, 0, &usage) < 0) return 2;
    fprintf(stderr, "%ld %d\n", usage.ru_maxrss, WIFEXITED(status) ? WEXITSTATUS(status) : -1);
    return 0;
}
