The script (exp-201's, unchanged; the second argument names the base tool):

"""Derived from a compare_tools artifact: the final head's own change against each peer.

The harness pairs every competitor with the anchor, not with each other. Two competitors'
timed samples at the same ordinal ran within one rotation of the anchor of each other, so
this pairs them at equal ordinals, exactly as the harness pairs variants, and reports the
peer's wall against the final head's as (peer - final) / final with the harness's own
deterministic percentile bootstrap of the median (2000 resamples, seed 0x5EED). The
anchor-normalized variant divides each sample by its adjacent anchor run first, which
removes whatever the host was doing to that rotation.
"""
import json, random, statistics, sys

def bootstrap(values, resamples=2000, seed=0x5EED):
    g = random.Random(seed); n = len(values); meds = []
    for _ in range(resamples):
        meds.append(statistics.median([values[g.randrange(n)] for _ in range(n)]))
    meds.sort()
    return meds[int(0.025 * (resamples - 1))], meds[int(0.975 * (resamples - 1))]

d = json.load(open(sys.argv[1]))
base = sys.argv[2] if len(sys.argv) > 2 else "fdu-final"
timed = [s for s in d["samples"] if not s["warmup"] and s.get("valid", True)]
wall = {}   # (tool, pair, ordinal) -> wall_ns
for s in timed:
    wall[(s["tool"], s["pair"], s["ordinal"])] = s["metrics"]["wall_ns"]
anchor = d["anchor"] if isinstance(d["anchor"], str) else d["anchor"]["name"]
ordinals = sorted({s["ordinal"] for s in timed})
def rows(normalize):
    out = []
    for peer in d["competitor_order"]:
        if peer == base: continue
        ratios = []
        for o in ordinals:
            try:
                b = wall[(base, base, o)]; p = wall[(peer, peer, o)]
                if normalize:
                    b /= wall[(anchor, base, o)]; p /= wall[(anchor, peer, o)]
            except KeyError:
                continue
            ratios.append((p - b) / b)
        m = statistics.median(ratios); lo, hi = bootstrap(ratios)
        out.append(f"  {peer:12s} n={len(ratios):2d}  {m*100:+.1f}% [{lo*100:+.1f}%, {hi*100:+.1f}%]")
    return out
print(f"{d['tree']['label'] if isinstance(d.get('tree'), dict) and 'label' in d['tree'] else sys.argv[1]}: peers against {base}, same-ordinal pairs, (peer - {base}) / {base}")
print("\n".join(rows(False)))
print("anchor-normalized (each sample over its adjacent anchor run):")
print("\n".join(rows(True)))
